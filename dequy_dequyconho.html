Số lượt gọi F(n): 0
Sẵn sàng mô phỏng đệ quy có nhớ.
ll F(int n)
{
    if(n==0) return 0;
    if(n==1) return 1;
    if(dp[n]>0) return dp[n];
    dp[n] = F(n-1) + F(n-2);
    return dp[n];
}
